# Stage 1: Builder - Installs Python dependencies
FROM python:3.11-slim as builder

# Set the working directory
WORKDIR /app

# Install build dependencies required to process pyproject.toml
# setuptools and wheel are needed to handle the project installation
RUN pip install --no-cache-dir pip setuptools wheel

# Copy the project definition file
COPY pyproject.toml .

# Install all project dependencies (including mcp, fastapi, uvicorn)
# The '.' installs the packages defined in the [project] dependencies table
RUN pip install --no-cache-dir .

# Stage 2: Final Image - A small runtime image
FROM python:3.11-slim

# Set the working directory
WORKDIR /app

# Copy the entire source code (which is now small because we excluded the large DB files)
COPY . /app/

# Copy the installed Python packages from the builder stage
# This ensures a clean runtime image without the build tools
COPY --from=builder /usr/local/lib/python3.11/site-packages /usr/local/lib/python3.11/site-packages

# Set the PYTHONPATH so Python can find your modules (like Source.MCP.Aqueduct_Server)
ENV PYTHONPATH=/app

# Expose the port the FastAPI/Uvicorn server listens on
EXPOSE 8080

# The startup command: Uvicorn runs your FastAPI application
# Format: uvicorn <module_path>:<app_instance_name>
CMD ["uvicorn", "Source.MCP.Aqueduct_Server:app", "--host", "0.0.0.0", "--port", "8080"]
